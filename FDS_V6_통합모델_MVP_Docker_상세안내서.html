def make_engine():
    host = os.getenv('DB_HOST', '127.0.0.1')
    port = int(os.getenv('DB_PORT', '3306'))
    user = os.getenv('DB_USER', 'root')
    password = os.getenv('DB_PASSWORD') or getpass('MySQL password: ')

    url = URL.create('mysql+pymysql', username=user, password=password, host=host, port=port, database=SCHEMA, query={'charset': 'utf8mb4'})

    return create_engine(url, pool_pre_ping=True, pool_recycle=1800)

2. V6 FDS 테이블 Grain과 필수 컬럼 확인 cell 5

① 작업 목적

모델이 기대하는 테이블과 컬럼이 DB에 실제로 존재하는지 사전 점검합니다.

② 왜 필요한가?

강의 환경이나 DB 버전이 바뀌면 컬럼명이 하나만 달라도 훨씬 뒤에서 KeyError가 발생할 수 있습니다. 초기에 Data Contract를 검사하면 원인을 빠르게 찾을 수 있습니다.

③ 코드에서 꼭 볼 부분

  • EXPECTED_COLUMNS는 테이블별 “필수 계약”입니다.
  • SELECT * ... LIMIT 0은 실제 행을 읽지 않고 컬럼 구조만 확인합니다.
  • actual과 expected의 차집합으로 누락 컬럼을 찾습니다.
  • status가 CHECK이면 즉시 RuntimeError로 중단합니다.

④ 실행 결과를 어떻게 읽나?

테이블별 존재 여부, 기대 컬럼 수, 실제 컬럼 수, 누락 컬럼, OK/CHECK 상태가 표로 표시됩니다.

⑤ 초보자 주의점

Grain은 “한 행이 무엇을 의미하는가”입니다. transactions는 거래 1건, customers는 고객 1명처럼 Grain이 다르므로 merge 시 many_to_one 관계가 중요합니다.

3. AS-OF DATE와 분석기간 cell 7

① 작업 목적

DB의 최신 거래시점을 기준일로 정하고 최근 12개월 구간을 계산합니다.

② 왜 필요한가?

컴퓨터의 오늘 날짜를 기준으로 자르면 재실행할 때 데이터 범위가 달라질 수 있습니다. 데이터 자체의 최신 시점을 AS-OF DATE로 잡으면 실습 재현성이 좋아집니다.

③ 코드에서 꼭 볼 부분

  • transactions와 card_payments의 MAX 시간 중 더 최신 값을 사용합니다.
  • DateOffset(months=LOOKBACK_MONTHS)로 시작일을 계산합니다.
  • 이후 SQL WHERE 조건에서 같은 start/as_of 파라미터를 사용합니다.

④ 실행 결과를 어떻게 읽나?

두 원천의 max_date와 최종 AS_OF_DATE, START_DATE가 출력됩니다.

⑤ 초보자 주의점

AS_OF_DATE 이후의 미래 데이터를 Feature 생성에 섞지 않는 것이 Data Leakage 방지의 첫 단계입니다.

실제 코드 보기 — Notebook cell 7
latest_dates = pd.read_sql(text("""
        SELECT 'transactions' AS source, MAX(transaction_at) AS max_date
        FROM transactions
        UNION ALL
        SELECT 'card_payments', MAX(payment_at)
        FROM card_payments
    """), engine)
latest_dates['max_date'] = pd.to_datetime(latest_dates['max_date'], errors='coerce')
AS_OF_DATE = latest_dates['max_date'].max()
START_DATE = AS_OF_DATE - pd.DateOffset(months=LOOKBACK_MONTHS)

display(latest_dates)
print('AS_OF_DATE:', AS_OF_DATE)
print('START_DATE:', START_DATE)

② 왜 필요한가?

⑤ 초보자 주의점

rolling을 잘못 쓰면 현재 행 또는 미래 행이 포함될 수 있습니다. 이 구현은 searchsorted와 prefix를 이용해 현재 이전 범위만 계산하려는 설계입니다.

실제 코드 보기 — Notebook cell 42
def add_point_in_time_history_features(df):
    # 메모리 절약을 위해 원본 DataFrame 자체를 정렬하고 Feature를 추가합니다.
    df.sort_values(['customer_id', 'event_at', 'event_uid'], inplace=True)
    df.reset_index(drop=True, inplace=True)

    work = df
    n = len(work)

    count_10m = np.zeros(n, dtype=np.int32)
    count_1h = np.zeros(n, dtype=np.int32)
    sum_1h = np.zeros(n, dtype=np.float64)
    count_24h = np.zeros(n, dtype=np.int32)
    sum_24h = np.zeros(n, dtype=np.float64)
    count_30d = np.zeros(n, dtype=np.int32)
    mean_30d = np.full(n, np.nan)
    std_30d = np.full(n, np.nan)
    minutes_prev = np.full(n, np.nan)

    W10 = int(pd.Timedelta(minutes=10).value)
    W1H = int(pd.Timedelta(hours=1).value)
    W24 = int(pd.Timedelta(hours=24).value)
    W30D = int(pd.Timedelta(days=30).value)

    started = time.perf_counter()

    for _, pos in work.groupby('customer_id', sort=False).indices.items():
        pos = np.asarray(pos, dtype=np.int64)

        times = (work.iloc[pos]['event_at'].astype('int64').to_numpy())

        amounts = (pd.to_numeric(work.iloc[pos]['amount_abs'], errors='coerce').fillna(0).to_numpy(float))

        right = np.arange(len(pos), dtype=np.int64)

        left10 = np.searchsorted(times, times - W10, side='left')
        left1h = np.searchsorted(times, times - W1H, side='left')
        left24 = np.searchsorted(times, times - W24, side='left')
        left30 = np.searchsorted(times, times - W30D, side='left')

        prefix = np.r_[0.0, np.cumsum(amounts),]
        prefix_sq = np.r_[0.0, np.cumsum(amounts ** 2),]

        count_10m[pos] = right - left10
        count_1h[pos] = right - left1h
        sum_1h[pos] = (prefix[right] - prefix[left1h])

        count_24h[pos] = right - left24
        sum_24h[pos] = (prefix[right] - prefix[left24])

        c30 = right - left30
        s30 = (prefix[right] - prefix[left30])
        sq30 = (prefix_sq[right] - prefix_sq[left30])

        count_30d[pos] = c30

        valid_mean = c30 > 0
        m30 = np.full(len(pos), np.nan)
        m30[valid_mean] = (s30[valid_mean] / c30[valid_mean])
        mean_30d[pos] = m30

        valid_std = c30 > 1
        var = np.full(len(pos), np.nan)
        var[valid_std] = (sq30[valid_std] - (s30[valid_std] ** 2) / c30[valid_std]) / (c30[valid_std] - 1)

        var = np.where(np.isnan(var), np.nan, np.maximum(var, 0))
        std_30d[pos] = np.sqrt(var)

        if len(pos) > 1:
            minutes_prev[pos[1:]] = (times[1:] - times[:-1]) / (60 * 1_000_000_000)

    work['customer_event_count_10m'] = count_10m
    work['customer_event_count_1h'] = count_1h
    work['customer_amount_sum_1h'] = sum_1h
    work['customer_event_count_24h'] = count_24h
    work['customer_amount_sum_24h'] = sum_24h
    work['customer_event_count_30d'] = count_30d
    work['customer_amount_mean_30d'] = mean_30d
    work['customer_amount_std_30d'] = std_30d
    work['minutes_since_prev_event'] = minutes_prev

    work['amount_vs_customer_mean_30d'] = np.where(work['customer_amount_mean_30d'].gt(0), work['amount_abs'] / work['customer_amount_mean_30d'], np.nan)

    work['amount_zscore_30d'] = np.where(work['customer_amount_std_30d'].gt(0), (work['amount_abs'] - work['customer_amount_mean_30d']) / work['customer_amount_std_30d'], np.nan)

    work['customer_history_available'] = (work['customer_event_count_30d'] >= 3).astype(int)

    print('Point-in-Time Feature 완료:', f'{time.perf_counter() - started:,.1f}초')

    return work

    yield ('R01_AMOUNT_ANOMALY', (df['amount_zscore_30d'].ge(RULE_CONFIG['large_amount_zscore']) & df['amount_abs'].ge(RULE_CONFIG['large_amount_min'])))

    yield ('R02_VELOCITY_10M', df['customer_event_count_10m'].ge(RULE_CONFIG['velocity_10m_count']))

    yield ('R03_HIGH_SUM_1H', (df['customer_amount_sum_1h'].add(df['amount_abs'], fill_value=0).ge(RULE_CONFIG['velocity_1h_sum'])))

    yield ('R04_NEW_DEVICE_HIGH_AMOUNT', (df['is_new_device_event'].fillna(0).eq(1) & df['amount_abs'].ge(RULE_CONFIG['new_device_amount'])))

    yield ('R05_CARD_REMOTE_RISK', (df['source_is_card'].eq(1) & df['is_online'].fillna(0).eq(1) & (df['foreign_ip_flag'].eq(1) | df['is_new_device_event'].fillna(0).eq(1))))

27-1. sklearn 입력 dtype/결측 표준화 cell 61

① 작업 목적

Pandas dtype과 pd.NA를 scikit-learn이 안정적으로 처리할 형태로 맞춥니다.

② 왜 필요한가?

nullable Int64, string, pd.NA가 섞이면 일부 Transformer/모델에서 dtype 오류가 날 수 있습니다.

③ 코드에서 꼭 볼 부분

  • 수치형은 float64와 np.nan으로 통일합니다.
  • 범주형은 object로 바꾸고 pd.NA를 np.nan으로 바꿉니다.
  • Train/Validation/Test 모두 같은 함수 prepare_model_input을 사용합니다.

④ 실행 결과를 어떻게 읽나?

X_train/X_valid/X_test와 y_train/y_valid/y_test가 준비됩니다.

⑤ 초보자 주의점

이 함수는 Feature Engineering이 아니라 “형식 표준화”입니다. 새로운 정보를 만들지 않습니다.

실제 코드 보기 — Notebook cell 61
# -----------------------------------------------------------------------------
# scikit-learn 전처리용 dtype / 결측값 표준화
# -----------------------------------------------------------------------------

def prepare_model_input(df):
    X = df.copy()

    # 수치형:
    # 숫자로 변환할 수 없는 값은 np.nan으로 변환하고
    # float64로 통일합니다.
    for col in NUMERIC_FEATURES:
        if col in X.columns:
            X[col] = (pd.to_numeric(X[col], errors="coerce").astype("float64"))

    # 범주형:
    # pandas의 pd.NA를 sklearn이 안정적으로 처리하는
    # np.nan으로 변환합니다.
    for col in CATEGORICAL_FEATURES:
        if col in X.columns:
            s = X[col].astype("object")
            X[col] = s.where(pd.notna(s), np.nan)

    return X

def best_f2_threshold(y_true, score):
    precision, recall, thresholds = precision_recall_curve(y_true, score)

    p = precision[:-1]
    r = recall[:-1]

    if len(thresholds) == 0:
        raise RuntimeError('Threshold를 계산할 수 없습니다.')

    f2 = (5 * p * r / (4 * p + r + 1e-12))

    table = pd.DataFrame({'threshold': thresholds, 'precision': p, 'recall': r, 'f2': f2,})

    idx = int(np.nanargmax(f2))

    return {'threshold': float(thresholds[idx]), 'precision': float(p[idx]), 'recall': float(r[idx]), 'f2': float(f2[idx]),}, table

def make_model_input(row):
    X = pd.DataFrame([{feature: row.get(feature, np.nan) for feature in MODEL_FEATURES}])
    for col in NUMERIC_FEATURES:
        if col in X.columns: X[col] = pd.to_numeric(X[col], errors='coerce').astype('float64')
    for col in CATEGORICAL_FEATURES:
        if col in X.columns:
            s = X[col].astype('object'); X[col] = s.where(pd.notna(s), np.nan)
    return X

def decide_action(score):
    if score >= FRAUD_ACTION_THRESHOLDS['block']: return 'BLOCK_AND_REVIEW'
    if score >= FRAUD_ACTION_THRESHOLDS['temporary_hold']: return 'TEMP_HOLD'
    if score >= FRAUD_ACTION_THRESHOLDS['step_up']: return 'STEP_UP_AUTH'
    return 'PASS'

def safe_number(row, key, default=0.0): return float(pd.to_numeric(pd.Series([row.get(key, default)]), errors='coerce').fillna(default).iloc[0])

Serving과 성능평가가 다른 이유

모델 개발 단계

후보 모델을 비교하고 Validation으로 Threshold와 Weight를 정합니다. 여기서는 “어떤 모델/정책을 쓸지 결정”합니다.

Serving 단계

이미 확정된 모델/정책으로 들어오는 Event를 판단합니다. 운영 중 임의로 Threshold를 바꾸지 않습니다.

현재 MVP의 정확한 의미: 원시 거래가 들어와 Feature를 실시간 계산하는 시스템이 아니라, 03에서 Feature Engineering이 끝난 Test 대표 행을 서비스처럼 재생합니다.

def num(row, key, default=0.0):
    return float(pd.to_numeric(pd.Series([row.get(key, default)]), errors='coerce').fillna(default).iloc[0])


def predict_event(row):
    started = time.perf_counter()

10. Dockerfile.streamlit 한 줄씩 해설

FROM python:3.11-slim — Python 3.11이 설치된 가벼운 Linux Image를 출발점으로 사용합니다.

ENV PYTHONDONTWRITEBYTECODE=1 — .pyc 파일 생성을 줄입니다. PYTHONUNBUFFERED=1은 로그가 버퍼에 오래 머물지 않고 바로 보이게 합니다.

WORKDIR /app — 이후 명령의 기본 작업폴더를 Container 내부 /app으로 설정합니다.

apt-get ... libgomp1 — LightGBM/XGBoost 같은 native library가 사용할 OpenMP 런타임을 설치합니다.

COPY requirements... + pip install — 먼저 의존성을 설치합니다. 코드가 바뀌어도 requirements가 같으면 Docker cache를 활용하기 좋은 순서입니다.

RUN mkdir -p /app/models /app/output — 모델과 Demo data 폴더를 만듭니다.

COPY ... — 앱과 03이 만든 Artifact를 Image 안으로 복사합니다.

EXPOSE 8501 — 이 Image의 서비스 포트가 8501임을 문서화합니다.

HEALTHCHECK — Streamlit의 /_stcore/health 주소가 응답하는지 주기적으로 확인합니다.

CMD [...] — Container가 시작될 때 Streamlit을 0.0.0.0:8501로 실행합니다. 0.0.0.0이어야 Container 바깥에서 접근할 수 있습니다.

11. docker-compose.local.yml 해설

Compose는 긴 docker build/docker run 옵션을 YAML 파일에 저장해 두는 도구입니다. 여기서는 fds-mvp 한 서비스를 정의합니다.

  • image: fds-v6-local:latest — 만들어질 Image 이름
  • build.context: . — 현재 프로젝트 폴더를 Docker build context로 사용
  • dockerfile: Dockerfile.streamlit — 일반 Dockerfile 대신 이 파일 사용
  • container_name — 실행 Container 이름 고정
  • init: true — PID 1 신호/좀비 프로세스 처리를 돕는 작은 init 사용
  • ports: 8501:8501 — PC의 8501을 Container의 8501로 연결
  • restart: unless-stopped — 사용자가 직접 중지한 경우가 아니면 재시작 정책 적용
Compose 전체 보기
name: fds-v6-local

services:
  fds-mvp:
    image: fds-v6-local:latest
    build:
      context: .
      dockerfile: Dockerfile.streamlit
    container_name: fds-v6-local
    init: true
    ports:
      - "8501:8501"
    restart: unless-stopped

모델 Serving 개념을 학습하기에는 충분하지만, “새 원시 거래가 즉시 들어오는 운영 시스템”은 아닙니다.

실제 운영으로 확장

신규 원시 거래 / API / Kafka
  ↓
Online Feature Engineering / Feature Store
  ↓
최근 10분·1시간·24시간·30일 이력 조회
  ↓
Rule Engine에서 Rule Score 즉시 계산
  ↓
ML Serving
  ↓
Hybrid Decision
  ↓
승인 / 추가인증 / 보류 / 차단검토
  ↓
결과 로그·모니터링·재학습

특히 Point-in-Time Feature를 운영에서도 같은 정의로 계산해야 학습-서비스 불일치가 생기지 않습니다.

가장 중요한 차이: 현재 Demo CSV의 rule_score와 History Feature는 03에서 이미 계산되어 있습니다. 실제 운영에서는 거래가 들어오는 순간 이 값들을 계산해 모델 앞단에 제공해야 합니다.

16. 초보자를 위한 성능지표 해석

지표쉽게 말하면FDS에서 보는 이유
Precision사기라고 잡은 것 중 진짜 사기 비율낮으면 정상 고객을 너무 많이 막습니다.
Recall실제 사기 중 잡아낸 비율낮으면 사기를 많이 놓칩니다.
F1Precision과 Recall 균형둘을 같은 중요도로 볼 때 유용합니다.
F2Recall을 더 중요하게 본 균형점수이 프로젝트에서 Threshold 선택 기준으로 사용합니다.
ROC-AUC정상과 사기를 순위로 분리하는 전체 능력보조 모델 비교 지표입니다.
PR-AUCPrecision-Recall 관계를 전체 Threshold에서 요약Fraud처럼 positive가 매우 적은 문제에서 모델 선택에 중요합니다.
Alert Rate전체 거래 중 경고로 보내는 비율운영 인력/고객마찰 비용과 연결됩니다.
예: 실제 Fraud 100건 중 90건을 잡으면 Recall 90%입니다. 그런데 Alert 900건 중 진짜 Fraud가 90건이면 Precision은 10%입니다. 즉 많이 잡지만 정상거래도 많이 경고하는 모델입니다. FDS에서는 이 두 비용의 균형이 중요합니다.

def make_model_input(row):
    X = pd.DataFrame([{feature: row.get(feature, np.nan) for feature in MODEL_FEATURES}])

    for col in NUMERIC_FEATURES:
        if col in X.columns:
            X[col] = pd.to_numeric(X[col], errors='coerce').astype('float64')

    for col in CATEGORICAL_FEATURES:
        if col in X.columns:
            s = X[col].astype('object')
            X[col] = s.where(pd.notna(s), np.nan)

    return X


def decide_action(score):
    if score >= FRAUD_ACTION_THRESHOLDS['block']:
        return 'BLOCK_AND_REVIEW'
    if score >= FRAUD_ACTION_THRESHOLDS['temporary_hold']:
        return 'TEMP_HOLD'
    if score >= FRAUD_ACTION_THRESHOLDS['step_up']:
        return 'STEP_UP_AUTH'
    return 'PASS'


def num(row, key, default=0.0):
    return float(pd.to_numeric(pd.Series([row.get(key, default)]), errors='coerce').fillna(default).iloc[0])


def predict_event(row):
    started = time.perf_counter()

    ml_risk_score = float(model.predict_proba(make_model_input(row))[:, 1][0])
    rule_score = num(row, 'rule_score', 0)
    rule_probability = float(np.clip(rule_score / 100, 0, 1))
    hybrid_score = ML_WEIGHT * ml_risk_score + RULE_WEIGHT * rule_probability

    return {
        'event_uid': str(row.get('event_uid', '')),
        'event_at': row.get('event_at'),
        'source_table': str(row.get('source_table', '')),
        'event_subtype': str(row.get('event_subtype', '')),
        'amount_abs': num(row, 'amount_abs', 0),
        'rule_score': rule_score,
        'ml_risk_score': ml_risk_score,
        'hybrid_score': float(hybrid_score),
        'alert': int(hybrid_score >= HYBRID_THRESHOLD),
        'action': decide_action(hybrid_score),
        'actual_label': int(num(row, 'fraud_label', 0)),
        'actual_fraud_type': str(row.get('fraud_type', 'NONE')),
        'is_new_device': int(num(row, 'is_new_device_event', 0)),
        'foreign_ip': int(num(row, 'foreign_ip_flag', 0)),
        'new_beneficiary': int(num(row, 'beneficiary_is_new', 0)),
        'beneficiary_risk': num(row, 'beneficiary_risk_score', 0),
        'balance_drain_ratio': num(row, 'balance_drain_ratio', 0),
        'latency_ms': (time.perf_counter() - started) * 1000,
    }
